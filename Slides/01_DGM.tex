\section{Objetivo de un DGM}
\begin{frame}
	\frametitle{Distribución de los datos}

	Los datos de alta dimensionalidad con los que trabajemos siguien una distribución de probabilidad:
	\begin{equation*}
		p_{\text{data}}
	\end{equation*}\pause

	\begin{itemize}
		\item Las combinaciones reales de los datos tendrán alta probabilidad, mientras que las combinaciones irreales (ruido, etc.) tendrán probabilidad prácticamente nula.
		\pause
		\item Esta distribución es \emph{desconocida} (tan solo tenemos acceso a una muestra de datos).
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Distribución del modelo}
	
	El DGM emplea una DNN con parámetros $\phi$ para obtener una distribución de probabilidad
	\begin{equation*}
		p_{\phi}
	\end{equation*}
	de manera que $p_{\phi} \approx p_{\text{data}}$.\pause
	\begin{alertblock}{Objetivo de un DGM}
		Aprender la distribución $p_{\text{data}}$ a partir de una muestra de datos, para poder generar nuevas muestras que sigan la misma distribución.
	\end{alertblock}
\end{frame}

\begin{frame}
	\frametitle{Generación de nuevas muestras}
	
	Una vez aprendido el modelo, podemos generar nuevas muestras de datos siguiendo la distribución $p_{\phi}$:
	\begin{equation*}
		x \sim p_{\phi} \approx p_{\text{data}}
	\end{equation*}
	
	\pause
	Esta generación debe cumplir:\pause
	\begin{itemize}
		\item Generación realista (Test de Lovelance)\pause
		\item Generación controlada
	\end{itemize}
\end{frame}


\section{Función de pérdida}

\begin{frame}
	\frametitle{Función de pérdida}
	
	\begin{definitionblock}{Divergencia}
		Una divergencia entre dos distribuciones de probabilidad $p_{\text{data}}$ y $p_{\phi}$ es una función $\cc{D}(p_{\text{data}}, p_{\phi})$ que cumple:
		\begin{itemize}
			\item No negatividad: $\cc{D}(p_{\text{data}}, p_{\phi}) \geq 0$
			\item Identidad: $\cc{D}(p_{\text{data}}, p_{\phi}) = 0 \iff p_{\text{data}} = p_{\phi}$ c.p.d.
		\end{itemize}
	\end{definitionblock}
	\pause

	Entrenar la DNN consiste en obtener los parámetros $\phi$ que minimicen la función de pérdida. Hay varias funciones de pérdida que se pueden emplear.
\end{frame}

\begin{frame}
	\frametitle{Divergencia KL}
	\begin{definitionblock}{Divergencia KL Directa}
		La divergencia KL directa entre dos distribuciones de probabilidad $p_{\text{data}}$ y $p_{\phi}$ es:
		\begin{equation*}
			\cc{D}_{\text{KL}}(p_{\text{data}} || p_{\phi}) := \int p_{\text{data}}(x) \log \frac{p_{\text{data}}(x)}{p_{\phi}(x)} dx
		\end{equation*}
		donde se ha tomado la primera distribución $p_{\text{data}}$ como referencia.
	\end{definitionblock}
	\pause

	Emplear la divergencia KL fuerza la \alert{cobertura del modelo}; es decir, toda muestra $x$ con $p_{\text{data}}(x) > 0$ debe tener $p_{\phi}(x) > 0$ (para no dividir entre cero).
	\begin{equation*}
		p_{\text{data}}(x) > 0 \implies p_{\phi}(x) > 0\qquad \forall x
	\end{equation*}
\end{frame}

\begin{frame}
	\frametitle{Divergencia KL}
	Puesto que no conocemos la distribución $p_{\text{data}}$, no podemos calcular 	de forma directa la divergencia KL.
	\begin{align*}
		\cc{D}_{\text{KL}}(p_{\text{data}} || p_{\phi})
		&:= \int p_{\text{data}}(x) \log \frac{p_{\text{data}}(x)}{p_{\phi}(x)} dx\\
		&= \bb{E}_{x \sim p_{\text{data}}} \left[ \log \frac{p_{\text{data}}(x)}{p_{\phi}(x)} \right]\\
		&= \bb{E}_{x \sim p_{\text{data}}} \left[ \log p_{\text{data}}(x) - \log p_{\phi}(x) \right]\\
		&= \bb{E}_{x \sim p_{\text{data}}} \left[ \log p_{\text{data}}(x) \right] - \bb{E}_{x \sim p_{\text{data}}} \left[ \log p_{\phi}(x) \right]
	\end{align*}
\end{frame}

\begin{frame}
	\frametitle{Divergencia KL}

	\begin{definitionblock}{Entropía}
		La entropía de una distribución de probabilidad $p_{\text{data}}$ es:
		\begin{equation*}
			\cc{H}(p_{\text{data}}) := -\int p_{\text{data}}(x) \log p_{\text{data}}(x) dx
			= -\bb{E}_{x \sim p_{\text{data}}} \left[ \log p_{\text{data}}(x) \right]
		\end{equation*}
	\end{definitionblock}
	\pause
	\begin{block}{Cálculo de la divergencia KL}
		\begin{equation*}
			\cc{D}_{\text{KL}}(p_{\text{data}} || p_{\phi})
			= -\cc{H}(p_{\text{data}}) - \bb{E}_{x \sim p_{\text{data}}} \left[ \log p_{\phi}(x) \right]
		\end{equation*}
	\end{block}\pause

	\begin{center}
		\alert{¡$\cc{H}(p_{\text{data}})$ no depende de los parámetros $\phi$!}
	\end{center}
\end{frame}



\begin{frame}
	\frametitle{Minimización de la divergencia KL}

	\begin{lemma}[Minimizar KL $\Longleftrightarrow$ MLE]
		\begin{equation*}
			\min_{\phi} \cc{D}_{\text{KL}}(p_{\text{data}} || p_{\phi}) \Longleftrightarrow \max_{\phi} \bb{E}_{x \sim p_{\text{data}}} \left[ \log p_{\phi}(x) \right]
		\end{equation*}
	\end{lemma}

	\pause

	Estimando mediante \alert{Monte Carlo} la esperanza (y teniendo en cuenta que se prefiere \alert{minimizar} la función de pérdida), obtenemos la función de pérdida empleada en la práctica:
	\begin{equation*}
		\hat{\cc{L}}_{\text{MLE}}(\phi) = \alert{-\frac{1}{N}} \sum_{i=1}^{N} \log p_{\phi}(x_i) \qquad x_i \sim p_{\text{data}}
	\end{equation*}
\end{frame}


\begin{frame}
	\frametitle{Divergencia de Fisher}
	\begin{definitionblock}{Divergencia de Fisher}
		La divergencia de Fisher entre dos distribuciones de probabilidad $p_{\text{data}}$ y $p_{\phi}$ es:
		\begin{equation*}
			\cc{D}_{\text{F}}(p_{\text{data}} || p_{\phi}) := E_{x \sim p_{\text{data}}} \left[ \left\| \nabla_x \log p_{\text{data}}(x) - \nabla_x \log p_{\phi}(x) \right\|_2^2 \right]
		\end{equation*}
		donde se ha tomado la primera distribución $p_{\text{data}}$ como referencia.
	\end{definitionblock}
	\alert{¿Por qué se emplean los logaritmos? ¿No se podría calcular sin ellos?}
\end{frame}

\begin{frame}
	\frametitle{$f$-Divergencias}
	\begin{definitionblock}{$f$-Divergencia}
		Sea $f: \bb{R}^{+} \to \bb{R}$ una función convexa con $f(1) = 0$. La $f$-divergencia entre dos distribuciones de probabilidad $p$ y $q$ es:
		\begin{equation*}
			\cc{D}_f(p || q)
			:= \int q(x) f\left( \frac{p(x)}{q(x)} \right) dx
		\end{equation*}
	\end{definitionblock}
\end{frame}
\begin{frame}
	\frametitle{$f$-Divergencias}
	\begin{exampleblock}{Casos particulares}
		\begin{itemize}
			\item Divergencia KL directa $\cc{D}_{\text{KL}}(p || q)$:
			$ f(u) = u \log u $
			\item Divergencia de Jensen-Shannon $\cc{D}_{\text{JS}}(p || q)$:
			$$ f(u) = \frac{1}{2} \left[ u\log u - (u+1) \log\left(\frac{u+1}{2}\right)\right] $$
			\item Variación Total $\cc{D}_{\text{TV}}(p || q)$:
			$ f(u) = \frac{1}{2} |u-1| $
		\end{itemize}
	\end{exampleblock}\pause
		\begin{align*}
			\cc{D}_{\text{JS}}(p || q) &= \frac{1}{2} \cc{D}_{\text{KL}}\left(p || \frac{1}{2}(p+q)\right) + \frac{1}{2} \cc{D}_{\text{KL}}\left(q || \frac{1}{2}(p+q)\right),\\
			\cc{D}_{\text{TV}}(p, q) &= \frac{1}{2} \int |p - q| dx = \sup_{A \subset \bb{R}^D} |p(A) - q(A)|.
		\end{align*}
\end{frame}


\section{Camino a obtener $p_{\phi}$}

\begin{frame}
	\frametitle{Camino a obtener $p_{\phi}$}
	Para que la función $p_{\phi}$ buscada sea realmente una función de densidad de probabilidad, debe cumplir:
	\begin{enumerate}
		\item No negatividad:
		\begin{equation*}
			p_{\phi}(x) \geq 0 \qquad \forall x
		\end{equation*}
		\item Normalización:
		\begin{equation*}
			\int p_{\phi}(x) dx = 1
		\end{equation*}
	\end{enumerate}

	\pause

	La DNN puede producir $E_{\phi}(x) \in \bb{R}$ para cada entrada $x$, pero hemos de realizar varias transformaciones para obtener $p_{\phi}(x)$.
\end{frame}
\begin{frame}
	\frametitle{No negatividad}

	Para garantizar la no negatividad, podemos emplear cualquier función positiva a $E_{\phi}(x)$, como $|E_{\phi}(x)|$ o $(E_{\phi}(x))^2$.\vspace{1cm} \pause 
	
	En la práctica, se emplea la función exponencial:
	\begin{equation*}
		\tilde{p}_{\phi}(x) = \exp(E_{\phi}(x)) \geq 0 \qquad \forall x
	\end{equation*}
\end{frame}

\begin{frame}
	\frametitle{Normalización}
	Para garantizar la normalización, podemos dividir la función $\tilde{p}_{\phi}(x)$ entre su integral:
	\begin{equation*}
		p_{\phi}(x) = \dfrac{\tilde{p}_{\phi}(x)}{\displaystyle
		\int \tilde{p}_{\phi}(x') dx'} = \frac{\exp(E_{\phi}(x))}{Z({\phi})}
	\end{equation*}
	donde se ha definido la \emph{constante de normalización} o \emph{función de partición}:
	\begin{equation*}
		Z({\phi}) := \int \tilde{p}_{\phi}(x') dx' = \int \exp(E_{\phi}(x')) dx'
	\end{equation*}
	
	\pause
	Esta integral es muy costosa de calcular, por lo que distintos tipos de DGM emplean distintas técnicas para aproximarla o evitarla.
\end{frame}

\section{DGM destacados}

\begin{frame}
	\frametitle{DGM destacados}
	Distintos DGM emplean distintas técnicas para aproximar o evitar la integral de normalización, haciendo distintos trade-offs entre:\pause
	\begin{itemize}
		\item \underline{Tractabilidad}: qué tan fácil es calcular la función de densidad de probabilidad $p_{\phi}(x)$ para una muestra $x$ dada.\pause
		\item \underline{Expresividad}: qué tan bien puede aproximar la verdadera complejidad de los datos.\pause
		\item \underline{Eficiencia de entrenamiento}: qué tan rápido se puede entrenar el modelo.
	\end{itemize}
\end{frame}


\begin{frame}
	\frametitle{Modelos Basados en Energía (EBM)}
	\begin{definitionblock}{Modelo Basado en Energía}
		Un Modelo Basado en Energía (EBM) es un DGM que emplea una DNN para obtener una función de energía $E_{\phi}(x)$ \emph{(donde se asigna baja energía a los datos más probables)}, y define:
		\begin{equation*}
			p_{\phi}(x) = \frac{\exp(\alert{-}E_{\phi}(x))}{Z({\phi})}
			\qquad \text{con} \qquad Z({\phi}) := \int \exp(\alert{-}E_{\phi}(x')) dx'
		\end{equation*}
	\end{definitionblock}\vspace{0.5cm}
	
	\pause

	Los Modelos de Difusión generarán muestras usando \emph{el gradiente del logaritmo de la función de densidad de probabilidad}, evitando así calcular $Z({\phi})$.

	\begin{center}
		\alert{De aquí en adelante, estoy más perdido.}
	\end{center}

\end{frame}

\begin{frame}
	\frametitle{Modelos Autoregresivos (AR)}
	\begin{definitionblock}{Modelo Autoregresivo}
		Un Modelo Autoregresivo (AR) es un DGM que descompone la función de densidad de probabilidad $p_{\phi}(x)$ en una cadena de distribuciones condicionales:
		\begin{equation*}
			p_{\phi}(x) = \prod_{i=1}^{D} p_{\phi}(x_i | x_{<i})
		\end{equation*}
		donde $x=(x_1, x_2, \dots, x_D)$ y $x_{<i} = (x_1, x_2, \dots, x_{i-1})$.
	\end{definitionblock}\pause
	Cada distribución condicional $p_{\phi}(x_i | x_{<i})$ se modela mediante una DNN y se normaliza por construcción (empleando determinadas funciones de activación como \emph{softmax} o \emph{Gaussiana parametrizada}), evitando así calcular $Z({\phi})$.
	\begin{center}
		\alert{¿Esto no se puede hacer con el resto?}
	\end{center}
\end{frame}

\begin{frame}
	\frametitle{Autoencoders Variacionales (VAE)}
	\begin{definitionblock}{Autoencoder Variacional}
		Un Autoencoder Variacional (VAE) es un DGM que emplea variables latentes $z$ que representan estructuras de alto nivel de los datos.\pause Emplea dos DNN:
		\begin{itemize}
			\item Un \emph{encoder} $q_{\theta}(z|x)$ que aproxima la distribución posterior de las variables latentes dado un dato $x$.\pause
			\item Un \emph{decoder} $p_{\phi}(x|z)$ que reconstruye el dato $x$ a partir de las variables latentes $z$.
		\end{itemize}
	\end{definitionblock}
\end{frame}
\begin{frame}
	\frametitle{Autoencoders Variacionales (VAE)}

	En vez de maximizar la probabilidad logarítmica de los datos, el VAE maximiza una cota inferior de la misma, llamada \emph{Evidence Lower Bound (ELBO)}:
	\begin{equation*}
		\cc{L}_{\text{ELBO}}(\theta, \phi; x) = \bb{E}_{q_{\theta}(z|x)} [\log p_{\phi}(x|z)] - \cc{D}_{\text{KL}}(q_{\theta}(z|x) \parallel p_{\text{prior}}(z))
	\end{equation*}\pause
	\begin{itemize}
		\item El primer término representa la reconstrucción correcta del dato $x$ a partir de las variables latentes $z$.\pause
		\item El segundo término busca regularizar la distribución posterior de las variables latentes para que se acerque a una distribución previa simple (como una Gaussiana estándar), evitando así el \emph{overfitting}.
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Flujos Normales (NF)}
	\begin{definitionblock}{Flujo Normal}
		Un Flujo Normal (NF) es un DGM que transforma una distribución simple de variables latentes $z \sim p_{\text{prior}}(z)$ (como una Gaussiana estándar) en la distribución de los datos $x \sim p_{\phi}(x)$ mediante una función (o cadena de funciones) invertible y diferenciable $f_{\phi}$:
		\begin{equation*}
			x = f_{\phi}(z) \qquad \text{y} \qquad z = f_{\phi}^{-1}(x)
		\end{equation*}
		La función de densidad de probabilidad $p_{\phi}(x)$ se obtiene mediante el cambio de variable:
		\begin{equation*}
			p_{\phi}(x) = p_{\text{prior}}(z) \left| \det \left( \frac{\partial f_{\phi}^{-1}(x)}{\partial x} \right) \right|
		\end{equation*}
	\end{definitionblock}
\end{frame}

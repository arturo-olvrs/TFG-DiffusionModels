\section{Energy-Based Models}
\begin{frame}{Energy-Based Models}

    Dado un conjunto de datos $\mathcal{D} = \{x_1, x_2, \ldots, x_N\}\subset \bb{R}^D$, entrenar directamente $p_\phi:\bb{R}^D:\bb{R}^+_0$ función de densidad no es sencillo. Por tanto, se emplea una \emph{función de energía} $E_\phi:\bb{R}^D\to \bb{R}$, que asigna valores bajos de energía a datos reales, y valores altos a datos irreales.

    \pause
    La función de energía induce una distribución de probabilidad:
    \begin{equation*}
        p_\phi(x) = \frac{\exp(-E_\phi(x))}{Z_\phi}, \quad Z_\phi = \int_{\bb{R}^D} \exp(-E_\phi(x))\ dx
    \end{equation*}
\end{frame}

\begin{frame}{Entrenamiento: Versión naïve}
    Buscamos maximizar la log-verosimilitud de los datos:
    \begin{align*}
        \mathcal{L}(\phi) &= \bb{E}_{x\sim p_\text{data}}[\log p_\phi(x)] = \bb{E}_{x\sim p_\text{data}}[-E_\phi(x) - \log Z_\phi]
        = \\&=\bb{E}_{x\sim p_\text{data}}[-E_\phi(x)] - \log Z_\phi
    \end{align*}
    \pause
    \begin{center}
        \color{red} \textbf{Problema:} $\log Z_\phi$ es intractable.
    \end{center}
\end{frame}

\section{Función de Score}
\begin{frame}{Solución: Función de Score}
    \begin{definitionblock}{Función de Score}
        Sea $p:\bb{R}^D\to \bb{R}^+_0$ una función de densidad. La función de score de $p$ es:
        \Func{s_p}{\bb{R}^D}{\bb{R}^D}{x}{\nabla_x \log p(x)}
        \pause
        $s_p(x)$ apunta en la dirección de mayor probabilidad, y su magnitud indica la rapidez con la que cambia la probabilidad.
    \end{definitionblock}

    \pause
    Se notará $s(x)$ en vez de $s_p(x)$ cuando no haya ambigüedad.
\end{frame}

\begin{frame}{Ventajas de la Función de Score}
    \begin{itemize}
        \item \ul{Ventaja Práctica:} Muchas funciones de densidad se dan como funciones de densidad no normalizadas $\tilde{p}(x)$, y se normalizan:
        \begin{equation*}
            p(x) = \frac{\tilde{p}(x)}{Z}, \quad Z = \int_{\bb{R}^D} \tilde{p}(x)\ dx
        \end{equation*}
        \pause
        \begin{equation*}
            \nabla_x \log p(x) = \nabla_x \log \tilde{p}(x) - \cancelto{0}{\nabla_x \log Z} = \nabla_x \log \tilde{p}(x)
        \end{equation*}
        \pause
        {\color{red} Se evita la intractabilidad de el cálculo de la integral intractable $Z$.}


        \pause
        \item \ul{Ventaja Teórica:} Se puede reconstruir $p(x)$ a partir de $s(x)$:
        \begin{equation*}
            \log p(x) = \log p(x_0) + \int_{0}^{1}  s(x_0 + t(x-x_0))^t (x-x_0)\ dt\qquad \forall x\in \bb{R}^D
        \end{equation*}
        \pause
        {\color{red} Se mantiene la misma expresividad.}
    \end{itemize}
\end{frame}

\begin{frame}{Entrenamiento de un EBM usando Score Matching}
    En vez de minimizar $\cc{D}_\text{KL}(p_\text{data}\|p_\phi)$, se minimiza\footnote{No se llegará al mismo resultado} $\cc{D}_\text{F}(p_\text{data}\|p_\phi)$:
    \begin{align*}
        \cc{L}_\text{SM}(\phi) &= \frac{1}{2}\cc{D}_\text{F}(p_\text{data}\|p_\phi) = \frac{1}{2}\bb{E}_{x\sim p_\text{data}}[\|s_\phi(x) - s_\text{data}(x)\|_2^2] \\
        &= \frac{1}{2}\bb{E}_{x\sim p_\text{data}}[\|\nabla_x \log p_\phi(x) - \nabla_x \log p_\text{data}(x)\|_2^2] \\
        \uncover<2->{&= \bb{E}_{x\sim p_\text{data}}\left[-\tr(\nabla_x^2E_\phi(x)) + \frac{1}{2}\|\nabla_x E_\phi(x)\|_2^2\right] + C}
    \end{align*}
    \pause
    \begin{center}
        \color{red} \textbf{Problema:} $\nabla_x^2E_\phi(x)$ requiere $\cc{O}(D)$ pases de backpropagation, lo que es ineficiente para $D$ grande. Además, minimizar segundas derivadas es más costoso.
    \end{center}
    
\end{frame}

\section{Langevin Sampling}

\begin{frame}{Langevin Sampling en Tiempo Discreto}
    \begin{definitionblock}{Langevin Sampling en Tiempo Discreto}
        Sea $p:\bb{R}^D\to \bb{R}^+_0$ una función de densidad. El proceso de Langevin en tiempo discreto con paso $\eta>0$ es:
        \begin{equation*}
            x_{n+1} = x_n + \eta \nabla_x \log p(x_n) + \sqrt{2\eta}\ \epsilon_n, \quad \epsilon_n\sim \cc{N}(0, \mathbf{I})
        \end{equation*}
    \end{definitionblock}
    
    Si $p$ es la distribución de probabilidad de los datos, entonces $x_n\sim p$ para $n\to \infty$.\pause

    La función de score guía el proceso de Langevin hacia regiones de alta probabilidad, mientras que el término de ruido $\sqrt{2\eta}\ \epsilon_n$ permite escapar de mínimos locales.\pause

    \begin{equation*}
        x_{n+1} = x_n - \eta \nabla_x E_\phi(x_n) + \sqrt{2\eta}\ \epsilon_n, \quad \epsilon_n\sim \cc{N}(0, \mathbf{I})
    \end{equation*}
    
\end{frame}

\begin{frame}{Langevin Sampling en Tiempo Continuo}
    \begin{definitionblock}{Langevin Sampling en Tiempo Continuo}
        Sea $p:\bb{R}^D\to \bb{R}^+_0$ una función de densidad. El proceso de Langevin en tiempo continuo es:
        \begin{equation*}
            dx = \nabla_x \log p(x)\ dt + \sqrt{2}\ dW_t
        \end{equation*}
        donde $W(t)$ es un movimiento Browniano estándar en $\bb{R}^D$.
    \end{definitionblock}

    Si $p$ es la distribución de probabilidad de los datos, entonces $x_t\sim p$ para $t\to \infty$.\pause

    La función de score guía el proceso de Langevin hacia regiones de alta probabilidad, mientras que el término de ruido $\sqrt{2}\ dW(t)$ permite escapar de mínimos locales.\pause

    \begin{equation*}
        dx = -\nabla_x E_\phi(x)\ dt + \sqrt{2}\ dW(t)
    \end{equation*}
    
\end{frame}

\begin{frame}{Dificultades de Langevin Sampling}
    
    Aunque es una técnica útil para generar muestras de una distribución de probabilidad, tiene un principal problema:
    \begin{center}
        Es muy difícil y tarda mucho moverse entre regiones de alta probabilidad (modas) separadas por regiones de baja probabilidad.
    \end{center}
\end{frame}

\section{Score-Based Models}
\begin{frame}{Motivación de Score-Based Models}
    Si al muestrear tan solo se necesita la función de score, ¿por qué no entrenar directamente un modelo para aproximar la función de score de los datos, olvidando la función de energía? \pause
    \begin{equation*}
        s_\phi(x) \approx s_\text{data}(x)
    \end{equation*}\pause

    La función de pérdida a minimizar queda:
    \begin{align*}
        \cc{L}_\text{SM}(\phi) &= \frac{1}{2}\bb{E}_{x\sim p_\text{data}}[\|s_\phi(x) - s_\text{data}(x)\|_2^2] \\
        \uncover<2->{&= \bb{E}_{x\sim p_\text{data}}\left[\tr(\nabla_x s_\phi(x)) + \frac{1}{2}\|s_\phi(x)\|_2^2\right] + C}
    \end{align*}
\end{frame}

\begin{frame}{Intuición de la Función de Pérdida}
    \begin{itemize}
        \item \ul{Minimizar $\|s_\phi(x)\|_2^2$:} Se busca que el modelo prediga un vector de score pequeño para datos reales; es decir, que el modelo prediga que los datos reales son de alta probabilidad.\pause
        \item \ul{Minimizar $\tr(\nabla_x s_\phi(x))$:} Cuando $s_\phi(x)=\nabla_x \log p_\text{data}(x)$, en un punto estacionario $x^*$ con $s_\phi(x^*)=0$, se tiene que:
        \begin{equation*}
            \log p_\text{data}(x) \approx \log p_\text{data}(x^*) + \frac{1}{2}(x-x^*)^t \nabla_x^2 \log p_\text{data}(x^*) (x-x^*)
        \end{equation*}
        \pause
        Si $\nabla_x^2 \log p_\text{data}(x^*)$ es negativa definida, entonces $x^*$ es un máximo local de $\log p_\text{data}(x)$, y por tanto un punto de alta probabilidad.\pause

        Si es definida negativa, entonces todos los autovalores de $\nabla_x^2 \log p_\text{data}(x^*)$ son negativos, y por tanto $\tr(\nabla_x^2 \log p_\text{data}(x^*)) < 0$. Por tanto, minimizar $\tr(\nabla_x s_\phi(x))$ fomenta que sea definida negativa, \emph{aunque no garantiza que lo sea (podría tener autovalores positivos aunque la suma de los autovalores sea negativa)}.

    \end{itemize}
\end{frame}


\section{Sliced SM}
\begin{frame}{Sliced Score Matching}
    Calcular $\tr(\nabla_x s_\phi(x))$ tiene coste $\cc{O}(D)$, lo que es ineficiente para $D$ grande. \pause
    \begin{proposition}[Identidad de Hutchinson]
        Sean $A\in \cc{M}_{D\times D}(\bb{R})$ y $s\in \bb{R}^D$ fijos. Sea $u\in \bb{R}^D$ un vector aleatorio con $\bb{E}[u]=0$ y $\bb{E}[uu^t] = \mathbf{I}$. Entonces:
        \begin{equation*}
            \bb{E}[u^t A u] = \tr(A)\qquad \bb{E}\left[\left(u^ts\right)^2\right] = s^t s = \|s\|_2^2
        \end{equation*}\pause
    \end{proposition}
    Por ejemplo, $u\rightsquigarrow \cc{N}(0, \mathbf{I})$ o $u\rightsquigarrow \text{Rademacher}^D$.
\end{frame}

\begin{frame}{Sliced Score Matching}
    \begin{equation*}
        \cc{L}_\text{SSM}(\phi) = \cc{L}_\text{SM}(\phi) = \bb{E}_{x\sim p_\text{data}, u}\left[u^t\left(\nabla_x s_\phi(x)\right) u + \frac{1}{2}\left(u^ts_\phi(x)\right)^2\right]
    \end{equation*}

    \pause
    Se usa JVP/VJP y la media muestral ($K$ muestras) como estimador insesgado con varianza $\cc{O}\left(\nicefrac{1}{K}\right)$, se obtiene un coste $\cc{O}(K)$, que es mucho menor que $\cc{O}(D)$ para $K\ll D$.\pause\vspace{0.5cm}
    
    {\color{red} ¿Sliced?} Cada $u$ es una dirección aleatoria, y se proyecta el score en esa dirección. Se repite $K$ veces y se promedia.\pause\vspace{0.5cm}

    En la práctica, $K=1$ y se deja que el minibatch haga el promedio.
\end{frame}
